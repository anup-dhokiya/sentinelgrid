Zero-Downtime Integration Architecture

SentinelGrid inspects incoming requests in-line with a sub-millisecond latency budget (p99 < 0.4ms). All verified requests receive an unforgeable RFC 8032 Ed25519 origin seal, guaranteeing upstream backends that traffic traversed security inspection.

Option A: Next.js & Vercel Edge Middleware

EDGE ENGINE

Drop this into middleware.js at the root of your project with fail-open circuit breaking:

import { NextResponse } from 'next/server';

export async function middleware(request) {
  const clientIp = request.headers.get('x-forwarded-for')?.split(',')[0]?.trim() || '127.0.0.1';
  try {
    const res = await fetch('https://sentinelgrid-one.vercel.app/v1/inspect', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        website_id: 'site-production',
        method: request.method,
        path: request.nextUrl.pathname,
        client_ip: clientIp,
        headers: Object.fromEntries(request.headers)
      }),
      signal: AbortSignal.timeout(1200)
    });
    if (res.ok) {
      const decision = await res.json();
      if (decision.action === 'BLOCK') {
        return new NextResponse('<h1>403 Forbidden: Dropped by SentinelGrid WAAP</h1>', { status: 403 });
      }
    }
  } catch (err) {}
  return NextResponse.next();
}

Option B: Nginx Reverse Proxy (Any Stack)

IN-LINE SUBREQUEST

Use auth_request to validate incoming packets before routing to upstream services:

server {
    listen 80;
    server_name app.corp.internal;

    location / {
        auth_request /_sentinel_inspect;
        proxy_pass http://upstream_service;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    location = /_sentinel_inspect {
        internal;
        proxy_pass https://sentinelgrid-one.vercel.app/v1/inspect;
        proxy_pass_request_body on;
        proxy_set_header Content-Type "application/json";
    }

    error_page 403 = @sentinel_blocked;
    location @sentinel_blocked {
        default_type text/html;
        return 403 '<h1>403 Quarantined by SentinelGrid</h1>';
    }
}

Option C: Python ASGI (FastAPI / Starlette)

PYTHON SDK

Intercept requests directly in FastAPI, Django ASGI, or Starlette:

from fastapi import FastAPI, Request
from starlette.responses import HTMLResponse
import httpx

app = FastAPI()

@app.middleware("http")
async def sentinelgrid_waf(request: Request, call_next):
    try:
        async with httpx.AsyncClient(timeout=1.0) as client:
            resp = await client.post("https://sentinelgrid-one.vercel.app/v1/inspect", json={
                "website_id": "site-python",
                "method": request.method,
                "path": request.url.path,
                "client_ip": request.client.host if request.client else "127.0.0.1",
                "headers": dict(request.headers)
            })
            if resp.status_code == 200 and resp.json().get("action") == "BLOCK":
                return HTMLResponse("<h1>403 Forbidden: Intercepted by WAAP</h1>", status_code=403)
    except Exception:
        pass # Fail-open
    return await call_next(request)

Option D: Node.js & Express Middleware

EXPRESS / CONNECT

Drop-in Express middleware for API backends and microservices:

const express = require('express');
const axios = require('axios');
const app = express();

app.use(async (req, res, next) => {
  try {
    const verdict = await axios.post('https://sentinelgrid-one.vercel.app/v1/inspect', {
      website_id: 'site-express',
      method: req.method,
      path: req.path,
      client_ip: req.ip || req.connection.remoteAddress,
      headers: req.headers
    }, { timeout: 1000 });

    if (verdict.data.action === 'BLOCK') {
      return res.status(403).send('<h1>403 Forbidden: Dropped by SentinelGrid WAAP</h1>');
    }
  } catch (err) {}
  next();
});
